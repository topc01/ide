# R3 — Chat (Claude Code local) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Chat con Claude Code local en la sesión activa: arranque lazy, streaming por WS, chips de tool/edit, footer de tokens, permisos skip/normal inline, interrupt, resume.

**Architecture:** El backend envuelve el **Claude Agent SDK de Python** (`claude-agent-sdk`) en un `ClaudeDriver` que traduce los mensajes del SDK a eventos normalizados; una capability WS (`chat.*`) arranca el proceso lazy al abrir el chat, hace streaming de eventos por `chat.event.<sid>`, y rutea permisos. El frontend (shadcn/1a, patrón AI Elements) acumula los eventos en un store y los renderiza (burbujas + chips + footer + permission prompt + composer).

**Tech Stack:** Python 3.12 / FastAPI / `claude-agent-sdk` / pytest (backend); React 19 + Vite + TS + Tailwind/shadcn + Zustand + vitest/RTL (frontend).

## Global Constraints

- **Sin API key.** Se usa el login de Claude Code (suscripción) vía el Agent SDK. Nunca setear `ANTHROPIC_API_KEY` ni usar el SDK `anthropic` (Messages API).
- **Sin fallos silenciosos.** Todo error viaja como envelope `{type:"error",payload:{code,message}}`; el chat muestra el error, nunca datos viejos como vivos.
- **Arranque lazy.** El proceso `claude` arranca al abrir el chat (`chat.start`), no al crear la sesión.
- **Identidad 1a vía shadcn** (regla #1 de CLAUDE.md). Tests asertan comportamiento/roles/texto, no estilos computados.
- **TDD.** Test que falla → implementación mínima → test que pasa → commit. Un commit por tarea.
- **Binding del Agent SDK (verificado 2026-07-12):** paquete `claude-agent-sdk`; import `from claude_agent_sdk import ClaudeSDKClient, ClaudeAgentOptions, AssistantMessage, TextBlock, ToolUseBlock, ResultMessage, PermissionResultAllow, PermissionResultDeny`. El paquete importa sin el binario `claude` (el binario sólo se invoca al hacer `client.query`), así que los tests construyen objetos de mensaje del SDK sin lanzar procesos. Si algún constructor de dataclass difiere de lo que muestra un step, ajustá los campos a la firma instalada (no aflojes la aserción de comportamiento).

```python
from __future__ import annotations
from dataclasses import dataclass

@dataclass
class TextDelta:
    text: str

@dataclass
class ToolUse:
    id: str
    name: str
    detail: str

@dataclass
class Edit:
    id: str
    path: str
    added: int
    removed: int

@dataclass
class Usage:
    input_tokens: int
    output_tokens: int
    total_tokens: int

@dataclass
class Done:
    session_id: str
    is_error: bool

@dataclass
class PermissionRequest:
    id: str
    tool: str
    detail: str

@dataclass
class ChatError:
    code: str
    message: str

def to_payload(event: object) -> dict:
    if isinstance(event, TextDelta):
        return {"kind": "text_delta", "text": event.text}
    if isinstance(event, ToolUse):
        return {"kind": "tool_use", "id": event.id, "name": event.name, "detail": event.detail}
    if isinstance(event, Edit):
        return {"kind": "edit", "id": event.id, "path": event.path,
                "added": event.added, "removed": event.removed}
    if isinstance(event, Usage):
        return {"kind": "usage", "inputTokens": event.input_tokens,
                "outputTokens": event.output_tokens, "totalTokens": event.total_tokens}
    if isinstance(event, Done):
        return {"kind": "done", "sessionId": event.session_id, "isError": event.is_error}
    if isinstance(event, PermissionRequest):
        return {"kind": "permission_request", "id": event.id, "tool": event.tool, "detail": event.detail}
    if isinstance(event, ChatError):
        return {"kind": "error", "code": event.code, "message": event.message}
    raise TypeError(f"evento de chat desconocido: {type(event)!r}")
```

```python
# server/ide_server/chat/translate.py
from __future__ import annotations
from claude_agent_sdk import AssistantMessage, TextBlock, ToolUseBlock, ResultMessage
from ide_server.chat.events import TextDelta, ToolUse, Edit, Usage, Done

_EDIT_TOOLS = {"Write", "Edit", "MultiEdit"}

def _lines(s: str | None) -> int:
    return len(s.splitlines()) if s else 0

def _tool_detail(name: str, inp: dict) -> str:
    if name == "Bash":
        return str(inp.get("command", ""))
    for key in ("file_path", "path", "pattern", "url"):
        if key in inp:
            return str(inp[key])
    return ""

def _edit_event(block: ToolUseBlock) -> Edit:
    inp = block.input
    path = str(inp.get("file_path", ""))
    if block.name == "Write":
        return Edit(id=block.id, path=path, added=_lines(inp.get("content")), removed=0)
    return Edit(id=block.id, path=path,
                added=_lines(inp.get("new_string")), removed=_lines(inp.get("old_string")))

def translate(message: object) -> list:
    out: list = []
    if isinstance(message, AssistantMessage):
        for block in message.content:
            if isinstance(block, TextBlock):
                out.append(TextDelta(text=block.text))
            elif isinstance(block, ToolUseBlock):
                if block.name in _EDIT_TOOLS:
                    out.append(_edit_event(block))
                else:
                    out.append(ToolUse(id=block.id, name=block.name,
                                       detail=_tool_detail(block.name, block.input)))
        return out
    if isinstance(message, ResultMessage):
        usage = message.usage or {}
        it, ot = int(usage.get("input_tokens", 0)), int(usage.get("output_tokens", 0))
        out.append(Usage(input_tokens=it, output_tokens=ot, total_tokens=it + ot))
        out.append(Done(session_id=message.session_id, is_error=bool(message.is_error)))
        return out
    return out
```

```python
# server/ide_server/chat/permissions.py
from __future__ import annotations

_READ_ONLY_TOOLS = ["Read", "Grep", "Glob", "LS"]

def resolve_permission(role: dict | None, override: str | None) -> tuple[str, list[str] | None]:
    if override == "skip":
        return ("bypassPermissions", None)
    if override == "normal":
        return ("default", None)
    role = role or {}
    if role.get("read_only"):
        return ("default", list(_READ_ONLY_TOOLS))
    if role.get("skip"):
        return ("bypassPermissions", None)
    return ("default", None)
```

```python
# server/ide_server/chat/driver.py
from __future__ import annotations
from typing import Awaitable, Callable
from claude_agent_sdk import ClaudeSDKClient, ClaudeAgentOptions
from ide_server.chat.translate import translate
from ide_server.chat.events import Done

EventCb = Callable[[object], None]
PermissionCb = Callable[[str, dict], Awaitable[bool]] | None

class ClaudeDriver:
    def __init__(self, *, cwd: str, system_prompt: str, permission_mode: str,
                 allowed_tools: list[str] | None, on_event: EventCb,
                 on_permission: PermissionCb, resume: str | None = None,
                 client_factory=None) -> None:
        self._on_event = on_event
        self._on_permission = on_permission
        self._factory = client_factory or ClaudeSDKClient
        self.session_id: str | None = resume
        opts: dict = {"cwd": cwd, "system_prompt": system_prompt,
                      "permission_mode": permission_mode, "include_partial_messages": False}
        if allowed_tools is not None:
            opts["allowed_tools"] = allowed_tools
        if resume:
            opts["resume"] = resume
        self._client = self._factory(options=ClaudeAgentOptions(**opts))

    async def start(self) -> None:
        await self._client.__aenter__()

    async def send(self, text: str) -> None:
        await self._client.query(prompt=text)
        async for message in self._client.receive_response():
            for event in translate(message):
                if isinstance(event, Done):
                    self.session_id = event.session_id
                self._on_event(event)

    async def interrupt(self) -> None:
        await self._client.interrupt()

    async def close(self) -> None:
        await self._client.__aexit__(None, None, None)
```

Nota: en `driver.py` el `ClaudeAgentOptions` de arranque puede aceptar el callback `can_use_tool` (se cablea en Task 5); acá se deja el modo de permisos y `allowed_tools`.

```python
# server/ide_server/capabilities/chat.py
from __future__ import annotations
import asyncio
from pathlib import Path
from ide_server.plugins import Registry
from ide_server.chat.driver import ClaudeDriver
from ide_server.chat.permissions import resolve_permission
from ide_server.chat.events import to_payload

def _load_session_context(root: Path, session_id: str) -> dict:
    # Devuelve {cwd, system_prompt, role} de la sesión. Implementación mínima:
    # busca el nodo en sessions.json y arma cwd=worktreePath, system_prompt del rol.
    from ide_server.sessions_store import load_sessions, find_node
    tree = load_sessions(root)
    node = find_node(tree, session_id)
    if node is None:
        raise KeyError(session_id)
    cwd = node.get("worktreePath") or str(root)
    role = node.get("role")
    desc = node.get("description") or ""
    system_prompt = f"{desc}".strip() or "Sos un asistente de código."
    return {"cwd": cwd, "system_prompt": system_prompt, "role": {"name": role} if role else None}

def register(registry: Registry, root: Path, manager, driver_factory=None) -> None:
    factory = driver_factory or ClaudeDriver
    drivers: dict[str, object] = {}
    pending: dict[str, asyncio.Future] = {}

    def _emit(session_id: str, event: object) -> None:
        asyncio.create_task(manager.broadcast(f"chat.event.{session_id}", to_payload(event)))

    async def _await_permission(session_id: str, request_id: str) -> bool:
        loop = asyncio.get_event_loop()
        fut: asyncio.Future = loop.create_future()
        pending[request_id] = fut
        try:
            return await fut
        finally:
            pending.pop(request_id, None)

    async def start(payload: dict) -> dict:
        sid = payload["sessionId"]
        if sid in drivers:
            return {"started": True, "resumed": True}
        ctx = _load_session_context(root, sid)
        mode, allowed = resolve_permission(ctx["role"], payload.get("override"))
        driver = factory(
            cwd=ctx["cwd"], system_prompt=ctx["system_prompt"],
            permission_mode=mode, allowed_tools=allowed,
            on_event=lambda e, s=sid: _emit(s, e),
            on_permission=lambda req_id, s=sid: _await_permission(s, req_id),
        )
        await driver.start()
        drivers[sid] = driver
        return {"started": True, "resumed": False}

    async def send(payload: dict) -> dict:
        await drivers[payload["sessionId"]].send(payload["text"])
        return {"sent": True}

    async def interrupt(payload: dict) -> dict:
        await drivers[payload["sessionId"]].interrupt()
        return {"interrupted": True}

    async def permission(payload: dict) -> dict:
        fut = pending.get(payload["requestId"])
        if fut is not None and not fut.done():
            fut.set_result(bool(payload["allow"]))
        return {"resolved": True}

    registry.register("chat.start", start)
    registry.register("chat.send", send)
    registry.register("chat.interrupt", interrupt)
    registry.register("chat.permission", permission)
```

```ts
// ui/src/state/chat.ts
import { create } from "zustand";

export type Chip =
  | { kind: "tool"; name: string; detail: string }
  | { kind: "edit"; path: string; added: number; removed: number };
export interface Msg { role: "user" | "agent"; text: string; chips: Chip[]; tokens?: number }
interface Convo {
  messages: Msg[]; streaming: boolean;
  permission: { id: string; tool: string; detail: string } | null;
  error: { code: string; message: string } | null; starting: boolean;
}
interface ChatState {
  convos: Record<string, Convo>;
  applyEvent: (sid: string, e: any) => void;
  setStarting: (sid: string, v: boolean) => void;
  pushUser: (sid: string, text: string) => void;
}

const empty = (): Convo => ({ messages: [], streaming: false, permission: null, error: null, starting: false });

function lastAgent(c: Convo): Msg {
  const last = c.messages[c.messages.length - 1];
  if (last && last.role === "agent") return last;
  const m: Msg = { role: "agent", text: "", chips: [] };
  c.messages.push(m);
  return m;
}

export const useChat = create<ChatState>((set) => ({
  convos: {},
  setStarting: (sid, v) => set((s) => {
    const c = { ...(s.convos[sid] ?? empty()), starting: v };
    return { convos: { ...s.convos, [sid]: c } };
  }),
  pushUser: (sid, text) => set((s) => {
    const c = { ...(s.convos[sid] ?? empty()) };
    c.messages = [...c.messages, { role: "user", text, chips: [] }];
    c.streaming = true;
    return { convos: { ...s.convos, [sid]: c } };
  }),
  applyEvent: (sid, e) => set((s) => {
    const c: Convo = { ...(s.convos[sid] ?? empty()) };
    c.messages = c.messages.map((m) => ({ ...m, chips: [...m.chips] }));
    if (e.kind === "text_delta") { lastAgent(c).text += e.text; }
    else if (e.kind === "tool_use") { lastAgent(c).chips.push({ kind: "tool", name: e.name, detail: e.detail }); }
    else if (e.kind === "edit") { lastAgent(c).chips.push({ kind: "edit", path: e.path, added: e.added, removed: e.removed }); }
    else if (e.kind === "usage") { lastAgent(c).tokens = e.totalTokens; }
    else if (e.kind === "done") { c.streaming = false; }
    else if (e.kind === "permission_request") { c.permission = { id: e.id, tool: e.tool, detail: e.detail }; }
    else if (e.kind === "error") { c.error = { code: e.code, message: e.message }; c.streaming = false; }
    return { convos: { ...s.convos, [sid]: c } };
  }),
}));
```

```tsx
    <div className="flex items-center gap-2 border-t border-border p-2">
      <Input value={text} onChange={(e) => setText(e.target.value)}
        onKeyDown={(e) => { if (e.key === "Enter") submit(); }}
        placeholder="Escribí un mensaje…" className="flex-1" />
      {streaming ? (
        <Button variant="secondary" size="icon" aria-label="detener" onClick={onStop}>
          <Square size={16} />
        </Button>
      ) : (
        <Button size="icon" aria-label="enviar" onClick={submit}>
          <Send size={16} />
        </Button>
      )}
    </div>
  );
}
```

```tsx
// ui/src/main/ChatView.tsx
import { useEffect, useRef } from "react";
import { useChat } from "../state/chat";
import { useSessions } from "../state/sessions";
import { getWs } from "../transport/wsClient";
import { startChat, sendChat, interruptChat, permissionDecision, subscribeChat } from "../transport/chatChannel";
import { MessageBubble } from "../chat/MessageBubble";
import { Composer } from "../chat/Composer";
import { PermissionPrompt } from "../chat/PermissionPrompt";

export function ChatView() {
  const activeSessionId = useSessions((s) => s.activeSessionId);
  const { convos, applyEvent, setStarting, pushUser } = useChat();
  const scrollRef = useRef<HTMLDivElement>(null);
  const sid = activeSessionId;

  useEffect(() => {
    if (!sid) return;
    setStarting(sid, true);
    subscribeChat(getWs(), sid, (e) => applyEvent(sid, e));
    startChat(getWs(), sid).finally(() => setStarting(sid, false));
  }, [sid, applyEvent, setStarting]);

  useEffect(() => { scrollRef.current?.scrollTo(0, scrollRef.current.scrollHeight); });

  if (!sid) {
    return <div className="p-4 text-sm text-muted-foreground">Seleccioná una sesión para chatear.</div>;
  }
  const c = convos[sid];
  return (
    <div className="flex h-full flex-col">
      <div ref={scrollRef} className="min-h-0 flex-1 space-y-3 overflow-auto p-4">
        {c?.starting && (!c || c.messages.length === 0) && (
          <div className="text-sm text-muted-foreground">Arrancando Claude Code…</div>)}
        {c?.messages.map((m, i) => <MessageBubble key={i} msg={m} />)}
        {c?.error && (
          <div className="rounded-md border border-border bg-card p-2 text-xs text-muted-foreground">
            {c.error.message}</div>)}
      </div>
      {c?.permission && (
        <PermissionPrompt tool={c.permission.tool} detail={c.permission.detail}
          onDecide={(allow) => permissionDecision(getWs(), sid, c.permission!.id, allow)} />)}
      <Composer streaming={!!c?.streaming}
        onSend={(t) => { pushUser(sid, t); sendChat(getWs(), sid, t); }}
        onStop={() => interruptChat(getWs(), sid)} />
    </div>
  );
}
```
