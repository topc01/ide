```python
# server/ide_server/worktrees.py
from __future__ import annotations
import subprocess
from pathlib import Path


class WorktreeError(Exception):
    """Error de operación de git worktree (nunca silencioso: viaja como envelope)."""


class BranchOccupied(WorktreeError):
    def __init__(self, branch: str) -> None:
        super().__init__(f"branch ocupada: {branch}")
        self.branch = branch


def _git(repo: Path, *args: str) -> subprocess.CompletedProcess[str]:
    return subprocess.run(
        ["git", "-C", str(repo), *args], capture_output=True, text=True,
    )


def _occupied_branches(repo: Path) -> set[str]:
    """Branches checkouteadas en algún worktree (incluye el principal)."""
    out = _git(repo, "worktree", "list", "--porcelain")
    if out.returncode != 0:
        raise WorktreeError(out.stderr.strip() or "git worktree list falló")
    occupied: set[str] = set()
    for line in out.stdout.splitlines():
        if line.startswith("branch "):
            ref = line[len("branch "):].strip()       # p.ej. refs/heads/main
            occupied.add(ref.rsplit("/", 1)[-1])
    return occupied


def list_branches(repo: Path) -> list[dict]:
    out = _git(repo, "branch", "--format=%(refname:short)")
    if out.returncode != 0:
        raise WorktreeError(out.stderr.strip() or "git branch falló")
    occupied = _occupied_branches(repo)
    names = [n.strip() for n in out.stdout.splitlines() if n.strip()]
    return [{"name": n, "occupied": n in occupied} for n in names]


def branch_occupied(repo: Path, branch: str) -> bool:
    return branch in _occupied_branches(repo)
```

```python
def _branch_exists(repo: Path, branch: str) -> bool:
    out = _git(repo, "rev-parse", "--verify", "--quiet", f"refs/heads/{branch}")
    return out.returncode == 0


def add_worktree(repo: Path, path: Path, branch: str, access: str,
                 base: str | None = None) -> None:
    if access not in ("rw", "ro"):
        raise WorktreeError(f"access inválido: {access}")
    exists = _branch_exists(repo, branch)

    if access == "rw":
        if exists and branch_occupied(repo, branch):
            raise BranchOccupied(branch)
        if exists:
            args = ["worktree", "add", str(path), branch]
        else:
            args = ["worktree", "add", "-b", branch, str(path), base or "HEAD"]
    else:  # ro → detached, siempre permitido
        target = branch if exists else (base or "HEAD")
        args = ["worktree", "add", "--detach", str(path), target]

    out = _git(repo, *args)
    if out.returncode != 0:
        raise WorktreeError(out.stderr.strip() or "git worktree add falló")


def remove_worktree(repo: Path, path: Path, force: bool = False) -> None:
    args = ["worktree", "remove", str(path)]
    if force:
        args.append("--force")
    out = _git(repo, *args)
    if out.returncode != 0:
        raise WorktreeError(out.stderr.strip() or "git worktree remove falló")


def has_uncommitted_changes(worktree: Path) -> bool:
    out = _git(worktree, "status", "--porcelain")
    if out.returncode != 0:
        raise WorktreeError(out.stderr.strip() or "git status falló")
    return bool(out.stdout.strip())
```

```python
# server/ide_server/roles.py
from __future__ import annotations
from dataclasses import dataclass, replace
from pathlib import Path
from ide_server.config import load_workspace_config


@dataclass(frozen=True)
class Role:
    name: str
    model: str | None = None
    system_prompt: str | None = None
    read_only: bool = False
    tools: list[str] | None = None


PRESETS: dict[str, Role] = {
    "lector": Role("lector", model="haiku", read_only=True),
    "revisor": Role("revisor", system_prompt="Revisá el código y reportá problemas.", read_only=True),
    "arquitecto": Role("arquitecto", system_prompt="Diseñá la arquitectura.", read_only=False),
    "disenador": Role("disenador", system_prompt="Enfocate en frontend/UI.", read_only=False),
}


def load_roles(root: Path) -> dict[str, Role]:
    roles = dict(PRESETS)
    cfg = load_workspace_config(root).get("roles", {})
    for name, data in cfg.items():
        base = roles.get(name, Role(name))
        roles[name] = replace(
            base,
            name=name,
            model=data.get("model", base.model),
            system_prompt=data.get("system_prompt", base.system_prompt),
            read_only=data.get("read_only", base.read_only),
            tools=data.get("tools", base.tools),
        )
    return roles


def resolve_role(root: Path, name: str | None) -> Role | None:
    if name is None:
        return None
    return load_roles(root)[name]
```

```python
# server/ide_server/sessions_store.py
from __future__ import annotations
import json
import re
from pathlib import Path


def slugify(name: str) -> str:
    s = re.sub(r"[^a-z0-9]+", "-", name.strip().lower())
    return s.strip("-")


def _path(root: Path) -> Path:
    return root / ".ide" / "sessions.json"


def load_sessions(root: Path) -> list[dict]:
    path = _path(root)
    if not path.is_file():
        return []
    try:
        data = json.loads(path.read_text(encoding="utf-8"))
        return data.get("sessions", [])
    except (json.JSONDecodeError, OSError):
        # sin fallos silenciosos: respaldar y re-iniciar (el aviso lo emite la capacidad)
        path.replace(path.with_suffix(".json.bak"))
        return []


def save_sessions(root: Path, tree: list[dict]) -> None:
    _path(root).write_text(json.dumps({"sessions": tree}, indent=2), encoding="utf-8")


def find_node(tree: list[dict], node_id: str) -> dict | None:
    for node in tree:
        if node.get("id") == node_id:
            return node
        found = find_node(node.get("children", []), node_id)
        if found is not None:
            return found
    return None


def insert_node(tree: list[dict], parent_id: str | None, node: dict) -> None:
    if parent_id is None:
        tree.append(node)
        return
    parent = find_node(tree, parent_id)
    if parent is None:
        raise KeyError(parent_id)
    parent.setdefault("children", []).append(node)


def rename_node(tree: list[dict], node_id: str, name: str) -> bool:
    node = find_node(tree, node_id)
    if node is None:
        return False
    node["name"] = name
    return True


def remove_node(tree: list[dict], node_id: str) -> bool:
    for i, node in enumerate(tree):
        if node.get("id") == node_id:
            del tree[i]
            return True
        if remove_node(node.get("children", []), node_id):
            return True
    return False
```

```python
# server/ide_server/capabilities/sessions.py
from __future__ import annotations
import uuid
from pathlib import Path
from ide_server.plugins import Registry
from ide_server.workspace import discover_repos
from ide_server.worktrees import (
    list_branches, branch_occupied, add_worktree, remove_worktree,
    has_uncommitted_changes,
)
from ide_server.sessions_store import (
    load_sessions, save_sessions, slugify, insert_node, rename_node,
    find_node, remove_node,
)
from ide_server.roles import resolve_role
from ide_server.launcher import NullLauncher, SessionLauncher


def register(registry: Registry, root: Path, launcher: SessionLauncher | None = None) -> None:
    launcher = launcher or NullLauncher()

    def _repo_path(repo_id: str) -> Path:
        for repo in discover_repos(root):
            if repo["id"] == repo_id:
                return root / repo["id"]
        raise KeyError(repo_id)

    async def tree(_payload: dict) -> dict:
        return {"sessions": load_sessions(root), "repos": discover_repos(root)}

    async def branches(payload: dict) -> dict:
        return {"branches": list_branches(_repo_path(payload["repoId"]))}

    async def create(payload: dict) -> dict:
        name = payload["name"]
        slug = slugify(name)
        node: dict = {
            "id": str(uuid.uuid4()), "name": name, "slug": slug,
            "description": payload.get("description"),
            "role": payload.get("role"),
            "status": "idle",
            "repoId": None, "branch": None, "access": None, "worktreePath": None,
            "expanded": True, "terminals": [], "children": [],
        }
        if payload.get("kind") == "work":
            repo = _repo_path(payload["repoId"])
            branch = payload.get("branch") or slug
            access = payload.get("access") or "rw"
            wt = root / ".ide" / "worktrees" / slug
            add_worktree(repo, wt, branch, access=access, base=payload.get("base"))
            node.update(repoId=payload["repoId"], branch=branch, access=access,
                        worktreePath=str(wt))

        tree_data = load_sessions(root)
        insert_node(tree_data, payload.get("parentId"), node)
        save_sessions(root, tree_data)
        launcher.launch(node, resolve_role(root, payload.get("role")))
        return {"session": node}

    async def rename(payload: dict) -> dict:
        tree_data = load_sessions(root)
        ok = rename_node(tree_data, payload["id"], payload["name"])
        if ok:
            save_sessions(root, tree_data)
        return {"renamed": ok}

    async def close(payload: dict) -> dict:
        tree_data = load_sessions(root)
        node = find_node(tree_data, payload["id"])
        if node is None:
            return {"closed": False}
        wt = node.get("worktreePath")
        if wt:
            repo = _repo_path(node["repoId"])
            if node.get("access") == "rw" and not payload.get("force") \
                    and has_uncommitted_changes(Path(wt)):
                raise UncommittedChanges(node["id"])
            remove_worktree(repo, Path(wt), force=bool(payload.get("force")))
        remove_node(tree_data, payload["id"])
        save_sessions(root, tree_data)
        return {"closed": True}

    for channel, handler in {
        "sessions.tree": tree, "sessions.branches": branches,
        "sessions.create": create, "sessions.rename": rename, "sessions.close": close,
    }.items():
        registry.register(channel, handler)


class UncommittedChanges(Exception):
    def __init__(self, session_id: str) -> None:
        super().__init__(f"cambios sin commitear en la sesión {session_id}")
        self.session_id = session_id
```

```ts
import type { WsClient } from "./ws";
import type { Repo } from "./envelope";

export interface SessionNode {
  id: string; name: string; slug?: string; status: "running" | "idle" | "done";
  role?: string | null; repoId?: string | null; branch?: string | null;
  access?: "rw" | "ro" | null; worktreePath?: string | null;
  expanded: boolean; children: SessionNode[];
}
export interface Branch { name: string; occupied: boolean; }

export interface CreatePayload {
  name: string; description?: string; role?: string | null; parentId?: string | null;
  kind: "coordinator" | "work"; repoId?: string; branch?: string; base?: string;
  access?: "rw" | "ro";
}

export function fetchTree(ws: WsClient): Promise<{ sessions: SessionNode[]; repos: Repo[] }> {
  return ws.request("sessions.tree", {});
}
export function createSession(ws: WsClient, payload: CreatePayload): Promise<{ session: SessionNode }> {
  return ws.request("sessions.create", payload);
}
export function renameSession(ws: WsClient, id: string, name: string): Promise<{ renamed: boolean }> {
  return ws.request("sessions.rename", { id, name });
}
export function closeSession(ws: WsClient, id: string, force = false): Promise<{ closed: boolean }> {
  return ws.request("sessions.close", { id, force });
```

```ts
// ui/src/state/sessions.ts
import { create } from "zustand";
import type { SessionNode } from "../transport/sessions";
import type { Repo } from "../transport/envelope";

function mapTree(nodes: SessionNode[], id: string, fn: (n: SessionNode) => SessionNode): SessionNode[] {
  return nodes.map((n) =>
    n.id === id ? fn(n) : { ...n, children: mapTree(n.children, id, fn) });
}

interface State {
  tree: SessionNode[]; repos: Repo[];
  activeSessionId: string | null; editingId: string | null;
  modalParentId: string | null | undefined;   // undefined = cerrado
  setData: (sessions: SessionNode[], repos: Repo[]) => void;
  setActive: (id: string) => void;
  toggleExpanded: (id: string) => void;
  startRename: (id: string) => void;
  stopRename: () => void;
  openModal: (parentId: string | null) => void;
  closeModal: () => void;
}

export const useSessions = create<State>((set) => ({
  tree: [], repos: [], activeSessionId: null, editingId: null, modalParentId: undefined,
  setData: (tree, repos) => set({ tree, repos }),
  setActive: (activeSessionId) => set({ activeSessionId }),
  toggleExpanded: (id) => set((s) => ({
    tree: mapTree(s.tree, id, (n) => ({ ...n, expanded: !n.expanded })) })),
  startRename: (editingId) => set({ editingId }),
  stopRename: () => set({ editingId: null }),
  openModal: (modalParentId) => set({ modalParentId }),
  closeModal: () => set({ modalParentId: undefined }),
}));
```
