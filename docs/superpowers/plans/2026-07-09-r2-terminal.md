# R2 — Terminal · Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** El módulo Terminal del rail + streaming de PTYs locales sobre R0/R1: abrir terminales on-demand en el `cwd` de la sesión (su worktree), verlas como pestañas con `xterm.js`, e interactuar en tiempo real (bytes bidireccionales, resize, ciclo de vida) — la fundación PTY que la rebanada de Orquestación/`fide` reusará.

**Architecture:** Un `PtyManager` (stdlib `pty.fork` + `select`, sin deps nuevas) gestiona el ciclo de vida de los PTYs. Una capacidad `terminals` (registrada en el `Registry` de R0) expone `terminal.open/write/resize/close/list`; la **salida** del PTY se transmite como `event` por WebSocket vía el `ConnectionManager.broadcast` de R0 (un `channel` por terminal id, datos en base64). El frontend monta `xterm.js` en una pestaña del panel principal, conectada a ese canal; el `TerminalPanel` del rail lista las terminales de la sesión activa y las abre.

**Tech Stack:** Python 3.12+ (stdlib `pty`/`select`/`fcntl`/`termios`), FastAPI, pytest, `uv`. React 18, Vite, TypeScript, `@xterm/xterm` + `@xterm/addon-fit`, Zustand, vitest + @testing-library/react.

Spec: `docs/superpowers/specs/2026-07-08-terminal-design.md` (incluye §Aclaraciones de planificación — alcance A).

## Global Constraints

- **Fidelidad de UI (no negociable):** dirección 1a. Terminal body `--terminal-bg` (`#0f0e0c`); header con `cwd` mono muted + chip de branch en accent. Tokens exactos de `docs/design/REFERENCES.md`. Nada de estética por defecto.
- **Sin fallos silenciosos:** fallo al spawnear el PTY (comando/cwd inválido) → envelope `{type:"error",payload:{code,message}}` tipado, renderizado en la pestaña. WS caído → el `TerminalTab` muestra "desconectado", no pinta output viejo como vivo.
- **Alcance A:** R2 = fundación de terminal genérica. NO arranca Claude Code como proceso de la sesión (el `SessionLauncher` sigue como `NullLauncher`). Claude-en-PTY + `fide` = rebanada siguiente que reusa el `PtyManager`.
- **PTY = stdlib, sin deps nuevas en el backend.** `pty.fork()` + `select` + `fcntl.ioctl(TIOCSWINSZ)` + `os.kill`.
- **Transporte:** un `channel` por terminal id sobre el envelope de R0. Salida → `event` en canal `terminal.data.<id>` (payload `{data: <base64>}`); salida de cierre → `event` `terminal.exit.<id>`. Entrada/resize/close/open/list → `req` (canales `terminal.write/resize/close/open/list`). Datos binarios en **base64** (JSON-safe).
- **Reusar R0/R1:** `Registry`/`handle_message`/`ConnectionManager.broadcast`, `discover_repos`, `load_sessions`/`find_node`/`save_sessions` (sessions_store), `SessionNode.terminals`, theming 1a, `WsClient`/`getWs`, `registerModule`, `useStore` (openTab/Tab type "terminal"), `useSessions` (activeSessionId/tree). El módulo "terminal" YA es un slot del rail (stub en `stubs.tsx`): R2 reemplaza su Panel.
- **Puertos dev:** backend `8787`, Vite `5173`. Backend `server/`, frontend `ui/`. Commits: conventional, atómicos, en español; uno por tarea.

```python
# server/ide_server/pty_manager.py
from __future__ import annotations
import os
import pty
import select
import signal
import uuid
from pathlib import Path


class PtyManager:
    def __init__(self) -> None:
        self._terms: dict[str, dict] = {}

    def spawn(self, cwd: Path, argv: list[str]) -> str:
        tid = uuid.uuid4().hex[:12]
        pid, fd = pty.fork()
        if pid == 0:                            # hijo: stdio ya es el slave pty
            try:
                os.chdir(str(cwd))
            except OSError:
                pass
            try:
                os.execvp(argv[0], argv)
            except OSError:
                os._exit(127)                   # execvp falló (comando inexistente)
        self._terms[tid] = {"pid": pid, "fd": fd}
        return tid

    def read(self, tid: str, timeout: float = 0.0) -> bytes:
        fd = self._terms[tid]["fd"]
        r, _, _ = select.select([fd], [], [], timeout)
        if not r:
            return b""
        try:
            return os.read(fd, 65536)
        except OSError:
            return b""                          # EOF / fd cerrado

    def write(self, tid: str, data: bytes) -> None:
        os.write(self._terms[tid]["fd"], data)

    def is_alive(self, tid: str) -> bool:
        t = self._terms.get(tid)
        if not t:
            return False
        try:
            wpid, _ = os.waitpid(t["pid"], os.WNOHANG)
            return wpid == 0
        except OSError:
            return False

    def close(self, tid: str) -> None:
        t = self._terms.pop(tid, None)
        if not t:
            return
        try:
            os.kill(t["pid"], signal.SIGKILL)
            os.waitpid(t["pid"], 0)             # reap: evita zombie
        except OSError:
            pass
        try:
            os.close(t["fd"])
        except OSError:
            pass

    def fd(self, tid: str) -> int:
        return self._terms[tid]["fd"]

    def list_ids(self) -> list[str]:
        return list(self._terms)
```

# server/ide_server/capabilities/terminals.py
from __future__ import annotations
import os
from pathlib import Path
from ide_server.plugins import Registry
from ide_server.pty_manager import PtyManager
from ide_server.workspace import discover_repos
from ide_server.sessions_store import load_sessions, find_node


def register(registry: Registry, root: Path, manager, pty: PtyManager | None = None) -> None:
    pty = pty or PtyManager()
    terms: dict[str, dict] = {}                 # tid -> {sessionId, cmd, cwd}

    def _repo_path(repo_id: str) -> Path:
        for repo in discover_repos(root):
            if repo["id"] == repo_id:
                return root / repo["id"]
        raise KeyError(repo_id)

    def _cwd(payload: dict) -> Path:
        if payload.get("sessionId"):
            node = find_node(load_sessions(root), payload["sessionId"])
            if node is None:
                raise KeyError(payload["sessionId"])
            return Path(node.get("worktreePath") or root)
        if payload.get("repoId"):
            return _repo_path(payload["repoId"])
        return root

    async def open_(payload: dict) -> dict:
        cwd = _cwd(payload)
        cmd = payload.get("cmd") or [os.environ.get("SHELL", "/bin/zsh")]
        tid = pty.spawn(cwd, cmd)
        terms[tid] = {"sessionId": payload.get("sessionId"), "cmd": cmd, "cwd": str(cwd)}
        return {"id": tid, "cwd": str(cwd)}

    async def list_(payload: dict) -> dict:
        sid = payload.get("sessionId")
        out = [{"id": tid, "cmd": t["cmd"], "alive": pty.is_alive(tid)}
               for tid, t in terms.items() if t["sessionId"] == sid]
        return {"terminals": out}

    registry.register("terminal.open", open_)
    registry.register("terminal.list", list_)

    # compartido con Tasks 4-5 (write/resize/close + streaming) vía cierre:
    register._pty = pty        # type: ignore[attr-defined]
    register._terms = terms    # type: ignore[attr-defined]
    register._manager = manager  # type: ignore[attr-defined]
```

> Nota: `write/resize/close` + el streaming se agregan dentro de este mismo `register` en Tasks 4-5 (reemplazan el cuerpo, como en R1 sessions).

                loop.create_task(manager.broadcast(
                    f"terminal.data.{tid}",
                    {"data": base64.b64encode(data).decode()}))
            if not pty.is_alive(tid):
                try:
                    loop.remove_reader(fd)
                except (OSError, ValueError):
                    pass
                loop.create_task(manager.broadcast(f"terminal.exit.{tid}", {}))

        loop.add_reader(fd, _on_readable)
        return {"id": tid, "cwd": str(cwd)}
```

// ui/src/transport/terminalChannel.ts
import type { WsClient } from "./ws";

export function openTerminal(ws: WsClient,
  opts: { sessionId?: string; repoId?: string; cmd?: string[] }): Promise<{ id: string; cwd: string }> {
  return ws.request("terminal.open", opts);
}
export function writeTerminal(ws: WsClient, id: string, data: string): Promise<any> {
  return ws.request("terminal.write", { id, data: btoa(data) });
}
export function resizeTerminal(ws: WsClient, id: string, cols: number, rows: number): Promise<any> {
  return ws.request("terminal.resize", { id, cols, rows });
}
export function closeTerminal(ws: WsClient, id: string): Promise<any> {
  return ws.request("terminal.close", { id });
}
export function subscribeTerminal(ws: WsClient, id: string,
  onData: (bytes: Uint8Array) => void, onExit: () => void): void {
  ws.on(`terminal.data.${id}`, (p: any) => {
    const bin = atob(p.data);
    const bytes = new Uint8Array(bin.length);
    for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i);
    onData(bytes);
  });
  ws.on(`terminal.exit.${id}`, () => onExit());
}

// ui/src/modules/terminal/TerminalPanel.tsx
import { useSessions } from "../../state/sessions";
import { useStore } from "../../state/store";

import type { SessionNode } from "../../transport/sessions";

function find(tree: SessionNode[], id: string | null): SessionNode | null {
  for (const n of tree) {
    if (n.id === id) return n;
    const f = find(n.children, id);
    if (f) return f;
  }
  return null;
}

import { useEffect, useRef } from "react";
import { Terminal } from "@xterm/xterm";
import { FitAddon } from "@xterm/addon-fit";
import "@xterm/xterm/css/xterm.css";
import { subscribeTerminal, writeTerminal, resizeTerminal } from "../transport/terminalChannel";
import { getWs } from "../transport/wsClient";

export function TerminalTab({ id }: { id: string }) {
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const term = new Terminal({
      fontSize: 12, fontFamily: '"JetBrains Mono", ui-monospace, monospace',
      theme: { background: "#0f0e0c", foreground: "#f0ece4", cursor: "#e0863f" },
    });
    const fit = new FitAddon();
    term.loadAddon(fit);
    if (ref.current) term.open(ref.current);
    try { fit.fit(); } catch { /* jsdom */ }
    resizeTerminal(getWs(), id, term.cols, term.rows);

    subscribeTerminal(getWs(), id,
      (bytes) => term.write(bytes),
      () => term.write("\r\n[proceso terminado]\r\n"));
    term.onData((d) => writeTerminal(getWs(), id, d));

    return () => term.dispose();
  }, [id]);

  return <div ref={ref} style={{ width: "100%", height: "100%", background: "var(--terminal-bg)" }} />;
}
