# Tasks (markdown local) Fase 1 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development
> (recomendado) o superpowers:executing-plans. Los steps usan checkbox (`- [ ]`).

**Goal:** El ítem "Tasks" del rail gestiona archivos `.md` en `.ide/tasks/`: listar, abrir,
crear y editar (guardado local debounced), sin API externa.

**Architecture:** Módulo nuevo `tasks` siguiendo la plantilla del módulo Notion: backend
`tasks/parse.py` (puro) + `tasks/store.py` + capability `tasks.list/read/save/create`; frontend
transport + store zustand + `TasksPanel` (rail) + `TaskTab` (editor markdown crudo con debounce).

**Tech Stack:** Python/FastAPI + React/TS/vitest. Tests: `cd server && uv run pytest`,
`cd ui && npm test`.

## Global Constraints

- **Sin fallos silenciosos:** errores (id inválido, archivo inexistente, IO) propagan → envelope
  en `bus.py`; el front muestra error, no éxito falso.
- **Guard de path:** `task_id` es un filename pelado (`[A-Za-z0-9._-]+`, sin `..`/`/`); si no,
  `InvalidTaskId`.
- **Convenciones:** título = primer `# H1` (fallback: id); status = primera línea `status: <x>`
  (opcional).
- **`registerModule` deduplica por id:** hay que QUITAR el `tasks` de `stubs.tsx`, si no el stub
  gana sobre el módulo real.
- **Commits atómicos, en español, conventional commits.** Un commit por tarea.
- **Fidelidad UI 1a ámbar/terminal:** título `#e8e6e1`, archivo mono `#75705f`, status/acciones
  ámbar `#e0863f`.

```python
from __future__ import annotations
import re

_H1 = re.compile(r"^#\s+(.+?)\s*$")
_STATUS = re.compile(r"^status:\s*(.+?)\s*$", re.IGNORECASE)


def parse_meta(markdown: str) -> dict:
    title = None
    status = None
    for line in markdown.splitlines():
        if title is None:
            m = _H1.match(line)
            if m:
                title = m.group(1)
        if status is None:
            m = _STATUS.match(line)
            if m:
                status = m.group(1)
    return {"title": title, "status": status}
```

```python
from __future__ import annotations
import re
from pathlib import Path
from ide_server.tasks.parse import parse_meta
from ide_server.sessions_store import slugify

_ID = re.compile(r"[A-Za-z0-9._-]+")


class InvalidTaskId(ValueError):
    pass


def _tasks_dir(root: Path) -> Path:
    d = root / ".ide" / "tasks"
    d.mkdir(parents=True, exist_ok=True)
    return d


def _task_path(root: Path, task_id: str) -> Path:
    if not task_id or ".." in task_id or not _ID.fullmatch(task_id):
        raise InvalidTaskId(task_id)
    return _tasks_dir(root) / f"{task_id}.md"


def _meta(task_id: str, markdown: str) -> dict:
    m = parse_meta(markdown)
    return {"id": task_id, "title": m["title"] or task_id, "status": m["status"]}


def list_tasks(root: Path) -> list[dict]:
    out = []
    for p in sorted(_tasks_dir(root).glob("*.md")):
        out.append(_meta(p.stem, p.read_text(encoding="utf-8")))
    return out


def read_task(root: Path, task_id: str) -> dict:
    path = _task_path(root, task_id)
    if not path.is_file():
        raise KeyError(task_id)
    markdown = path.read_text(encoding="utf-8")
    return {**_meta(task_id, markdown), "markdown": markdown}


def write_task(root: Path, task_id: str, markdown: str) -> dict:
    path = _task_path(root, task_id)
    if not path.is_file():
        raise KeyError(task_id)
    path.write_text(markdown, encoding="utf-8")
    return _meta(task_id, markdown)


def create_task(root: Path, name: str) -> dict:
    base = slugify(name) or "tarea"
    slug, n = base, 1
    while (_tasks_dir(root) / f"{slug}.md").exists():
        n += 1
        slug = f"{base}-{n}"
    markdown = f"# {name}\nstatus: todo\n"
    (_tasks_dir(root) / f"{slug}.md").write_text(markdown, encoding="utf-8")
    return {"id": slug, "title": name, "status": "todo"}
```

```python
from __future__ import annotations
from pathlib import Path
from ide_server.plugins import Registry
from ide_server.tasks import store


def register(registry: Registry, root: Path) -> None:
    async def list_(_p: dict) -> dict:
        return {"tasks": store.list_tasks(root)}

    async def read(p: dict) -> dict:
        return store.read_task(root, p["id"])

    async def save(p: dict) -> dict:
        return store.write_task(root, p["id"], p["markdown"])

    async def create(p: dict) -> dict:
        return {"task": store.create_task(root, p["name"])}

    for name, fn in [("list", list_), ("read", read), ("save", save), ("create", create)]:
        registry.register(f"tasks.{name}", fn)
```

```ts
import type { WsClient } from "./ws";

export interface TaskRef { id: string; title: string; status: string | null }
export interface TaskDoc extends TaskRef { markdown: string }
export type TaskMeta = TaskRef;

export async function listTasks(ws: WsClient): Promise<TaskRef[]> {
  return (await ws.request("tasks.list", {})).tasks;
}
export async function readTask(ws: WsClient, id: string): Promise<TaskDoc> {
  return ws.request("tasks.read", { id });
}
export async function saveTask(ws: WsClient, id: string, markdown: string): Promise<TaskMeta> {
  return ws.request("tasks.save", { id, markdown });
}
export async function createTask(ws: WsClient, name: string): Promise<TaskRef> {
  return (await ws.request("tasks.create", { name })).task;
}
```

```ts
import { create } from "zustand";
import type { TaskRef } from "../transport/tasksChannel";
import { listTasks, createTask } from "../transport/tasksChannel";
import { getWs } from "../transport/wsClient";

interface TasksState {
  tasks: TaskRef[]; loading: boolean; error: string | null;
  load: () => Promise<void>;
  create: (name: string) => Promise<TaskRef>;
}
const msg = (e: unknown) => (e as { message?: string })?.message ?? "desconectado";

export const useTasks = create<TasksState>((set) => ({
  tasks: [], loading: false, error: null,
  load: async () => {
    set({ loading: true });
    try { set({ tasks: await listTasks(getWs()), loading: false, error: null }); }
    catch (e) { set({ loading: false, error: msg(e) }); }
  },
  create: async (name) => {
    const ref = await createTask(getWs(), name);
    try { set({ tasks: await listTasks(getWs()), error: null }); } catch (e) { set({ error: msg(e) }); }
    return ref;
  },
}));
```

```tsx
import { useEffect, useState } from "react";
import { Plus, ListChecks } from "lucide-react";
import { useTasks } from "../../state/tasks";
import { useStore } from "../../state/store";

export function TasksPanel() {
  const { tasks, error, load, create } = useTasks();
  const openTab = useStore((s) => s.openTab);
  const [adding, setAdding] = useState(false);
  const [name, setName] = useState("");

  useEffect(() => { load(); }, [load]);

  function open(id: string, title: string) {
    openTab({ id: `task:${id}`, type: "task", label: title });
  }
  async function submit() {
    if (!name.trim()) return;
    const ref = await create(name.trim());
    setName(""); setAdding(false);
    if (ref) open(ref.id, ref.title);
  }

  return (
    <div className="p-3">
      <div className="mb-2 flex items-center justify-between">
        <span className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">Tareas</span>
        <button onClick={() => setAdding(true)} className="flex items-center gap-1 text-[11px] font-semibold text-[#e0863f]"><Plus size={12} />Nueva</button>
      </div>
      {adding && (
        <input autoFocus placeholder="Nombre de la tarea" value={name}
          onChange={(e) => setName(e.target.value)}
          onKeyDown={(e) => { if (e.key === "Enter") submit(); if (e.key === "Escape") setAdding(false); }}
          className="mb-2 w-full rounded border border-border bg-background px-2 py-1 font-mono text-[11.5px]" />
      )}
      {error && <div className="px-1 py-2 text-xs text-destructive">desconectado — {error}</div>}
      {tasks.length === 0 && !error && <div className="px-1 py-2 text-xs text-muted-foreground">todavía no hay tareas</div>}
```

```tsx
import { useEffect, useRef, useState } from "react";
import { readTask, saveTask } from "../transport/tasksChannel";
import { getWs } from "../transport/wsClient";
import { useTasks } from "../state/tasks";

type Status = "idle" | "saving" | "saved" | "error";

export function TaskTab({ id }: { id: string }) {
  const taskId = id.slice("task:".length);
  const [title, setTitle] = useState("");
  const [markdown, setMarkdown] = useState<string | null>(null);
  const [status, setStatus] = useState<Status>("idle");
  const [error, setError] = useState<string | null>(null);
  const reloadRail = useTasks((s) => s.load);
  const timer = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);

  useEffect(() => {
    let cancelled = false;
    readTask(getWs(), taskId)
      .then((d) => { if (!cancelled) { setTitle(d.title); setMarkdown(d.markdown); } })
      .catch((e) => { if (!cancelled) setError((e as { message?: string })?.message ?? "desconectado"); });
    return () => { cancelled = true; };
  }, [taskId]);

  function onChange(value: string) {
    setMarkdown(value);
    clearTimeout(timer.current);
    setStatus("saving");
    timer.current = setTimeout(async () => {
      try {
        const meta = await saveTask(getWs(), taskId, value);
        setTitle(meta.title); setStatus("saved");
        reloadRail();
      } catch (e) { setStatus("error"); setError((e as { message?: string })?.message ?? "error"); }
    }, 800);
  }

  if (error && markdown === null) return <div className="p-4 text-sm text-destructive">no se pudo abrir la tarea — {error}</div>;
  if (markdown === null) return <div className="p-4 text-sm text-muted-foreground">cargando…</div>;
  return (
    <div className="flex h-full flex-col">
      <div className="flex items-center gap-3 border-b border-border px-5 py-3">
        <span className="text-lg font-bold text-foreground">{title}</span>
        <span className="text-[10px] text-muted-foreground">
          {status === "saving" ? "guardando…" : status === "saved" ? "guardado" : status === "error" ? "error al guardar" : ""}
        </span>
      </div>
      <textarea
        value={markdown}
        onChange={(e) => onChange(e.target.value)}
        className="min-h-0 flex-1 resize-none bg-transparent px-5 py-4 font-mono text-[13px] leading-relaxed text-foreground outline-none"
      />
    </div>
  );
}
```

> el selector y devuelve `{ load }`, el llamado `useTasks((s)=>s.load)` recibe el selector — en el
> mock de arriba `useTasks: () => ({ load: vi.fn() })` IGNORA el selector y devuelve el objeto, así
> que `reloadRail` sería ese objeto, no la fn. Ajusta el mock del test para que respete el selector:
